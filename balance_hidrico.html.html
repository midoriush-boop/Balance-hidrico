  <script>

    function parseVal(id) {

      return parseFloat(document.getElementById(id).value) || 0;

    }

    function calcularBalance() {

      const peso = parseVal('peso');

      const horas = parseVal('horas');

      if (peso <= 0 || horas <= 0) {

        alert('Ingresa el peso del paciente y las horas del turno.');

        return;

      }

      const ingresos = parseVal('soluciones') + parseVal('medicacion') + parseVal('oral');

      const diuresis = parseVal('diuresis');

      const egresos = diuresis + parseVal('drenajes') + parseVal('otrosEgresos');

      const balanceNeto = ingresos - egresos;

      const ritmoDiuretico = diuresis / (peso * horas);

      document.getElementById('resIngresos').innerText = `${ingresos.toFixed(0)} mL`;

      document.getElementById('resEgresos').innerText = `${egresos.toFixed(0)} mL`;

     

      const resNeto = document.getElementById('resNeto');

      if (balanceNeto > 0) {

        resNeto.innerText = `+${balanceNeto.toFixed(0)} mL (Positivo)`;

        resNeto.style.color = '#38bdf8';

      } else if (balanceNeto < 0) {

        resNeto.innerText = `${balanceNeto.toFixed(0)} mL (Negativo)`;

        resNeto.style.color = '#f87171';

      } else {

        resNeto.innerText = `0 mL (Neutro)`;

        resNeto.style.color = '#94a3b8';

      }

      document.getElementById('resRitmo').innerText = `${ritmoDiuretico.toFixed(2)} mL/kg/h`;

      const tag = document.getElementById('tagRitmo');

      if (ritmoDiuretico < 0.5) {

        tag.className = 'tag tag-oligo';

        tag.innerText = 'OLIGURIA (< 0.5 mL/kg/h)';

      } else if (ritmoDiuretico <= 1.5) {

        tag.className = 'tag tag-normo';

        tag.innerText = 'DIURESIS NORMAL (0.5 - 1.5 mL/kg/h)';

      } else {

        tag.className = 'tag tag-poly';

        tag.innerText = 'POLIURIA (> 1.5 mL/kg/h)';

      }

      document.getElementById('resultado').style.display = 'block';

    }

    function limpiar() {

      document.querySelectorAll('input').forEach(input => {

        if (input.id === 'horas') input.value = '8';

        else input.value = '';

      });

      document.getElementById('resultado').style.display = 'none';

    }
